        return &API{K8sClientProvider: k8sClientProvider}

// KubeconfigAuthMiddleware validates the provided kubeconfig from the request header.
func KubeconfigAuthMiddleware() gin.HandlerFunc {
        return func(c *gin.Context) {
                kubeconfigB64 := c.GetHeader("X-Kubeconfig-B64")
                if kubeconfigB64 == "" {
                        c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{"error": "X-Kubeconfig-B64 header is required"})
                        return
                }

                kubeconfig, err := base64.StdEncoding.DecodeString(kubeconfigB64)
                if err != nil {
                        c.AbortWithStatusJSON(http.StatusBadRequest, gin.H{"error": "invalid base64 for kubeconfig"})
                        return
                }

                k8sClient, clientset, err := k8s.NewClientsFromKubeconfig(kubeconfig)
                if err != nil {
                        c.AbortWithStatusJSON(http.StatusUnauthorized, gin.H{"error": "failed to create k8s client", "details": err.Error()})
                        return
                }

                c.Set("k8sClient", k8sClient)
                c.Set("clientset", clientset)
                c.Next()
        }
}

// Connect handler validates the provided kubeconfig from the request header.
func Connect(c *gin.Context) {
        // The actual connection test is handled by the KubeconfigAuthMiddleware.
        // If we reach here, it means the client is valid.
        c.JSON(http.StatusOK, gin.H{"message": "kubeconfig is valid and connection successful"})
}

// handleK8sError checks the error from the kubernetes client and returns the appropriate HTTP status code.
func handleK8sError(c *gin.Context, context string, err error) {
        if k8serrors.IsInvalid(err) || k8serrors.IsBadRequest(err) {
                // Validation error (e.g., missing required fields)
                c.JSON(http.StatusBadRequest, gin.H{"error": context, "details": err.Error()})
        } else if k8serrors.IsAlreadyExists(err) {
                // Resource already exists
                c.JSON(http.StatusConflict, gin.H{"error": context, "details": err.Error()})
        } else if k8serrors.IsNotFound(err) {
                c.JSON(http.StatusNotFound, gin.H{"error": context, "details": err.Error()})
        } else {
                // Other errors are treated as internal server errors
                c.JSON(http.StatusInternalServerError, gin.H{"error": context, "details": err.Error()})
        }
}

// clientFromContext retrieves the kubernetes client from the gin context.
func clientFromContext(c *gin.Context) (client.Client, bool) {
        k8sClientVal, ok := c.Get("k8sClient")
        if !ok {
                c.JSON(http.StatusUnauthorized, gin.H{"error": "kubeconfig not provided or invalid"})
                return nil, false
        }
        k8sClient := k8sClientVal.(client.Client)
        return k8sClient, true
}

func clientsetFromContext(c *gin.Context) (kubernetes.Interface, bool) {
        clientsetVal, ok := c.Get("clientset")
        if !ok {
                c.JSON(http.StatusUnauthorized, gin.H{"error": "kubeconfig not provided or invalid"})
                return nil, false
        }
        clientset := clientsetVal.(kubernetes.Interface)
        return clientset, true
}

func GetLogs(c *gin.Context) {
        clientset, ok := clientsetFromContext(c)
        if !ok {
                return
        }

        namespace := c.Param("namespace")
        podName := c.Param("pod_name")
        containerName := c.Query("container")
        if containerName == "" {
                c.JSON(http.StatusBadRequest, gin.H{"error": "container query parameter is required"})
                return
        }
        tailLinesStr := c.DefaultQuery("tailLines", "1000")
        tailLines, err := strconv.ParseInt(tailLinesStr, 10, 64)
        if err != nil {
                c.JSON(http.StatusBadRequest, gin.H{"error": "invalid tailLines query parameter", "details": err.Error()})
                return
        }

        logs, err := k8s.GetPodLogs(clientset, namespace, podName, containerName, tailLines)
        if err != nil {
                handleK8sError(c, "failed to get pod logs", err)
                return
        }

        c.String(http.StatusOK, logs)
}
